Hình~\ref{fig:irms-context-overview} thể hiện \emph{context/process diagram}: IRMS ở trung tâm điều phối vận hành, các vai trò con người nằm ngoài ranh giới này.

\begin{itemize}
  \item \textbf{Các vai trò vận hành:} Gồm \textsf{Customer}, \textsf{Host / Reception}, \textsf{Server / Floor Staff}, \textsf{Kitchen Operations}, \textsf{Cashier}, \textsf{Manager} và \textsf{System Administrator}. Đây là những điểm tương tác và quyết định yêu cầu phản hồi của hệ thống.
  \item \textbf{Các vai trò không phải hệ thống ngoài}: các kênh thanh toán, thông báo và biên lai được xem là adapter biên nội bộ.
  \item \textbf{Các luồng nghiệp vụ:} như \enquote{reservation request}, \enquote{kitchen ticket}, \enquote{transaction state} hay \enquote{dấu vết kiểm toán}. Trách nhiệm của IRMS ở mức trao đổi dữ liệu, không chỉ ở mức vai trò sử dụng.
  \item \textbf{Ranh giới trách nhiệm}: IRMS chịu trách nhiệm điều phối đặt bàn, gọi món, bếp, hóa đơn, tồn kho và quản trị; các chi tiết như thanh toán, thông báo và biên lai được xử lý qua service hoặc adapter biên nội bộ.
\end{itemize}

Hình~\ref{fig:irms-usecase-overview} thể hiện sơ đồ use cases tổng quát. Các use cases được gom theo các cụm lớn bám theo nghiệp vụ. Gồm các khối \texttt{Reservation and Seating}, \texttt{Ordering and Kitchen}, \texttt{Billing and Settlement}, \texttt{Inventory and Replenishment Support} và \texttt{Operations and Governance}.

\paragraph{Các cụm chức năng.}
\begin{itemize}
  \item \textbf{Reservation and Seating}: Gồm đặt bàn, tiếp nhận khách, danh sách chờ, giữ chỗ và mở phiên phục vụ tại bàn. Thuộc service \textsf{Reservation \& Seating Service}.
  \item \textbf{Ordering and Kitchen}: Gồm tạo đơn, điều phối phiếu bếp, cập nhật tiến độ chế biến, xử lý ưu tiên và xác nhận món đã phục vụ. Thuộc service \textsf{Ordering \& Menu Service} và \textsf{Kitchen Coordination Service}.
  \item \textbf{Billing and Settlement}: Gồm tạo hóa đơn, tách hóa đơn, xử lý thanh toán, hoàn tiền và phát hành biên lai; Thuộc service \textsf{Billing \& Settlement Service}.
  \item \textbf{Inventory and Replenishment Support}: Gồm ghi nhận tiêu hao, cảnh báo mức tồn thấp và hỗ trợ quyết định nhập thêm nguyên liệu từ dữ liệu lịch sử. Thuộc service \textsf{Inventory Monitoring Service}.
  \item \textbf{Governance and Analytics}: Gồm quản trị thực đơn, phân quyền, kiểm toán, báo cáo và điều hành vận hành; đây là phạm vi của \textsf{Reporting Service} và \textsf{IAM \& Audit Service}.
\end{itemize}

Ở góc nhìn kiến trúc, sơ đồ tách rõ \textbf{đường giao dịch nóng} khỏi \textbf{đường hỗ trợ}. Các thao tác như xác nhận đơn, cập nhật trạng thái món sẵn sàng, ghi nhận thanh toán và đóng phiên bàn là các điểm phải phản hồi dứt khoát qua gọi đồng bộ giữa các biên phù hợp. Ngược lại, các tác vụ như làm mới báo cáo, phát cảnh báo mức tồn thấp, đồng bộ trạng thái phụ trợ hay gửi thông báo có thể đi trên tuyến bất đồng bộ theo hướng event-driven nếu điều đó không làm sai trạng thái phục vụ trước mặt khách hàng.

\subsection{Góc nhìn sử dụng}
\subsubsection{Kịch bản 1: Từ đặt bàn đến tiếp nhận và xếp bàn}
\begin{enumerate}
  \item Khách hàng hoặc lễ tân tạo đặt bàn.
  \item \textsf{Reservation \& Seating Service} kiểm tra khả dụng, gán bàn hoặc giữ chỗ và ghi nhận \textsf{ReservationCreated} nếu cần phát thông báo.
  \item Khi khách đến, \textsf{Host UI} xác nhận tiếp nhận và mở \textsf{TableSession}.
  \item Trạng thái bàn được cập nhật qua \textsf{TableStatusChanged} để tuyến phục vụ nhìn thấy bàn sẵn sàng nhận order.
\end{enumerate}

\subsubsection{Kịch bản 2: Từ gọi món đến điều phối bếp và phục vụ}
\begin{enumerate}
  \item Phục vụ tạo đơn gọi món và cấu hình tùy chọn món hoặc gợi ý dị ứng trên \textsf{Server POS UI}.
  \item \textsf{Ordering \& Menu Service} xác nhận đơn gọi món và phát \textsf{OrderPlaced}.
  \item \textsf{Kitchen Coordination Service} nhận order, tách phiếu theo trạm bếp và hiển thị lên \textsf{Kitchen Display UI}.
  \item Bếp cập nhật trạng thái từng dòng phiếu qua các mốc như \textsf{OrderItemStarted} và \textsf{OrderReady}; màn hình gọi món nhận lại trạng thái sẵn sàng.
  \item Phục vụ giao món và đánh dấu đã phục vụ; \textsf{Inventory Monitoring Service} nhận sự kiện liên quan để ghi nhận tiêu hao và cập nhật kho.
\end{enumerate}

\subsubsection{Kịch bản 3: Từ hóa đơn đến thanh toán và biên lai}
\begin{enumerate}
  \item Thu ngân hoặc phục vụ gom các đơn gọi món của bàn thành hóa đơn.
  \item \textsf{Billing \& Settlement Service} áp dụng khuyến mãi, tiền boa và tách hóa đơn nếu cần.
  \item Thanh toán được thực hiện qua tiền mặt hoặc qua \textsf{PaymentGateway}.
  \item Khi thanh toán đủ, hệ thống ghi nhận \textsf{PaymentCompleted}, phát hành biên lai và chuyển \textsf{TableSession} sang chuẩn bị đóng.
\end{enumerate}

\subsubsection{Kịch bản 4: Từ cảnh báo tồn kho thấp đến quyết định của quản lý}
\begin{enumerate}
  \item \textsf{Inventory Monitoring Service} cập nhật lượng tồn kho thực tế sau mỗi món đã phục vụ hoặc sau mỗi lần điều chỉnh thủ công.
  \item Nếu xuống dưới ngưỡng, hệ thống phát \textsf{StockLow}.
  \item \textsf{Manager Console} nhận cảnh báo, hiển thị nguyên liệu bị ảnh hưởng và món có nguy cơ hết hàng.
  \item Quản lý quyết định nhập thêm hàng hoặc tạm ẩn món trong menu.
\end{enumerate}

  \item \textbf{\textsf{Host UI} và \textsf{Reservation \& Seating Service}} chịu trách nhiệm kiểm tra khả dụng, giữ chỗ và xác nhận tiếp nhận khách.
  \item \textbf{\textsf{Reservation \& Seating Service}} là ranh giới nơi bàn được chuyển từ trạng thái chờ sang trạng thái đang phục vụ thông qua \textsf{TableSession}.
  \item \textbf{\textsf{Notification Service} và \textsf{Server POS UI}} chỉ được kích hoạt sau khi phiên bàn đã được mở thành công.
\end{itemize}

\paragraph{Ý nghĩa}
\begin{itemize}
  \item Phải có cơ chế chống mở trùng \textsf{TableSession} cho cùng một bàn hoặc cùng một lượt tiếp nhận.
  \item Trạng thái bàn và trạng thái đặt bàn không nên bị ép gộp thành một thực thể duy nhất, nếu không các chuyển tiếp quan trọng của vận hành sẽ trở nên khó khăn.
\end{itemize}

Giúp rút ngắn thời gian phản hồi tại màn hình gọi món và bảo vệ đơn đã chốt khi một thành phần phía sau như KDS hoặc đồng bộ trạng thái gặp trục trặc tạm thời.

  \item \textbf{\textsf{Menu Service}} lưu giữ trạng thái khả dụng và giá tại thời điểm xác nhận đơn, giúp tránh nhập nhằng khi menu thay đổi sau đó.
  \item \textbf{\textsf{Inventory Monitoring Service}} chỉ tham gia ở nhánh hỗ trợ, không chặn bước xác nhận đơn ở tuyến đầu.
\end{itemize}

\paragraph{Ý nghĩa}
\begin{itemize}
  \item Đơn gọi món, phiếu bếp và cập nhật trạng thái không nên bị ràng buộc trong cùng một bước đồng bộ dài.
  \item Cơ chế chống lặp, correlation ID và khả năng thử lại là bắt buộc nếu kiến trúc muốn vừa nhanh vừa không làm mất dữ liệu đơn.
\end{itemize}

\end{itemize}

\end{itemize}

\renewcommand{\arraystretch}{1.3}

\begin{longtable}{|p{4cm}|p{10cm}|}
\hline
\textbf{Use Case ID} & UC-RES-01 \\ \hline
\textbf{Use Case Name} & Tạo đặt bàn \\ \hline
\textbf{Primary Actors} & Khách hàng, Lễ tân \\ \hline

\textbf{Description} & Tạo một đặt bàn hợp lệ cho ngày, giờ và số lượng khách cụ thể. \\ \hline

\textbf{Trigger} & Khách hàng gửi yêu cầu đặt bàn qua quầy lễ tân, điện thoại hoặc web/app. \\ \hline

\textbf{Preconditions} &
\begin{itemize}[leftmargin=*, noitemsep]
  \item Nhà hàng có mở nhận đặt bàn trong khung giờ yêu cầu.
  \item Có tối thiểu tên và số điện thoại khách hàng.
  \item Sơ đồ bàn và năng lực phục vụ đã được cấu hình.
\end{itemize} \\ \hline

\textbf{Postconditions} &
